#pubblicare su github
git add .
git commit -m "Implementazioni varie"
git push origin main

#ricevere da github
git pull origin main

#***************** ripristinare lo stato di github
#Scarica le ultime novità da GitHub (senza applicarle ancora):
git fetch origin
#Ripristina tutti i file tracciati
git reset --hard origin/main
#Rimuovi file e cartelle non tracciati
git clean -fd

#Attenzione: I comandi reset --hard e clean -fd eliminano definitivamente tutte le modifiche locali non committate e non sarà possibile recuperarle. Dopo aver eseguito questi comandi, il tuo progetto sul PC sarà identico al 100% a quello presente su GitHub.

#*********pubblicare su VPS IONOS

# Compila e pubblica l'app nella cartella di produzione
dotnet publish /root/MuzzlebookRepo/Muzzlebook.csproj -c Release -o /var/www/muzzlebook

# Assegna i permessi all'utente del web server
chown -R www-data:www-data /var/www/muzzlebook

# Riavvia l'applicazione
systemctl restart muzzlebook.service

#***Scaricare  il db:
scp root@31.70.156.199:/var/www/muzzlebook/Data/muzzlebook.db C:\progetti\muzzlebook\Data\muzzlebook_dal_server.db
//poi rinominarlo in muzzlebook.db e sostituirlo a quello locale

#***cancellare il db:
systemctl stop muzzlebook.service
rm /var/www/muzzlebook/muzzlebook.db
systemctl start muzzlebook.service


#*********Aggiornare il db su VPS IONOS
#Ferma il servizio dell'app Sul terminale SSH del server
systemctl stop muzzlebook.service

#Invia il database con SCP
scp C:\progetti\muzzlebook\Data\muzzlebook.db root@31.70.156.199:/var/www/muzzlebook/Data/

#Ripristina i permessi e riavvia
